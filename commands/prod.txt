==========================================================================
 PRODUCTION — Docker commands
 Stack: Redis + backend (uvicorn, UVICORN_WORKERS from .env) + Celery
        worker (CELERY_CONCURRENCY from .env) + frontend + nginx gateway.
        Postgres = external AWS RDS, files = external S3 — no local db
        service (see docs/AWS_SETUP.md / docs/DEPLOYMENT.md).
 File:  docker-compose.prod.yml     Env: .env  (from .env.example + PROD profile)
 IMPORTANT: edit .env and replace every CHANGE_ME secret, plus DATABASE_URL
 (RDS endpoint) and S3_BUCKET/S3_REGION first.
==========================================================================

# ---- setup (once) ----
./scripts/create-env.sh PROD     # or interactive: ./scripts/create-env.sh
# then replace all CHANGE_ME values in .env, and fill in DATABASE_URL (RDS)
# + STORAGE_PROVIDER=s3 + S3_BUCKET/S3_REGION

# ---- start / stop (or use scripts/prod/start.sh | stop.sh) ----
docker compose -f docker-compose.prod.yml --env-file .env up --build -d
docker compose -f docker-compose.prod.yml down                # stop (Redis broker volume kept)
docker compose -f docker-compose.prod.yml down --volumes       # stop + drop the Redis volume too (queued jobs lost, not app data — RDS/S3 are untouched either way)

# ---- logs ----
docker compose -f docker-compose.prod.yml logs -f
docker compose -f docker-compose.prod.yml logs -f backend worker worker-extract

# ---- status / scale the worker ----
docker compose -f docker-compose.prod.yml ps
docker compose -f docker-compose.prod.yml up -d --scale worker-extract=3

# ---- deploy a new version (backend and/or frontend code changed) ----
# scripts/prod/start.sh is just `up --build -d`: it does NOT git pull and does
# NOT restart nginx. So, in order:
git pull origin <branch>                                  # 1. get the code first
bash scripts/prod/start.sh                                # 2. rebuilds changed images, recreates changed containers
docker compose -f docker-compose.prod.yml restart nginx   # 3. nginx caches upstream IPs; recreated containers get new ones
docker compose -f docker-compose.prod.yml ps              # 4. backend, worker (healthy = Celery answers ping), frontend, nginx up
# The frontend image is rebuilt too (that is how UI changes reach users), so a
# hard refresh (Ctrl+F5) may be needed in the browser afterwards.

# ---- rebuild + rolling update after a code change (manual equivalent) ----
docker compose -f docker-compose.prod.yml build
docker compose -f docker-compose.prod.yml up -d

# ---- changing .env (model names, secrets, limits) ----
# `restart` does NOT re-read .env — compose reads env_file only when a container
# is CREATED. Recreate instead:
docker compose -f docker-compose.prod.yml --env-file .env up -d backend worker worker-extract
docker compose -f docker-compose.prod.yml restart nginx   # backend got a new IP
# e.g. OPENAI_VISION_MODEL (all vision extraction, incl. bulk roster) and
# AGENT_CHAT_MODEL (Ask-AI chat) are independent settings — see docs/SYSTEM.md.

# ---- rebuild frontend ONLY (no backend changes) ----
docker compose -f docker-compose.prod.yml --env-file .env build frontend
docker compose -f docker-compose.prod.yml --env-file .env up -d frontend
docker compose -f docker-compose.prod.yml restart nginx

# ---- shells / migrations ----
docker compose -f docker-compose.prod.yml exec backend bash
docker compose -f docker-compose.prod.yml exec backend alembic upgrade head   # targets DATABASE_URL in .env

# ---- connect to RDS directly (reads DATABASE_URL out of .env) ----
psql "$(grep -E '^DATABASE_URL' .env | cut -d= -f2- | sed 's#+asyncpg##;s#?ssl=require#?sslmode=require#')"

# ---- database backup / restore (against RDS, not a local container) ----
# Prefer an RDS snapshot for anything you'd need to restore from (Console →
# RDS → your instance → Actions → Take snapshot, or automated backups are
# already on) — a plain pg_dump is fine for a quick ad-hoc export:
pg_dump "$(grep -E '^DATABASE_URL' .env | cut -d= -f2- | sed 's#+asyncpg##;s#?ssl=require#?sslmode=require#')" \
  > backup_$(date +%F).sql
psql "$(grep -E '^DATABASE_URL' .env | cut -d= -f2- | sed 's#+asyncpg##;s#?ssl=require#?sslmode=require#')" \
  < backup_$(date +%F).sql

# ---- health ----
curl -fsS http://localhost:8080/health   # nginx container -> backend (host nginx+certbot proxies 443 -> this)
# App (via host nginx/certbot): https://yourdomain.com
